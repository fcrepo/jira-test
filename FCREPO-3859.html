Description

The LDP specification indicates that any subject should allowed as statements in any LDP Resource (LDPR). This functionality isn't consistent between containers (which seem to work) and binary resources (which don't). In particular, binary resources seem to transfer all statements from a metadata record into the subject of the resource itself.

This example applies the same metadata, that includes a label for an subject term, to both a container and a binary resource:

However, the underlying ashley.art~fcr-desc.nt seems correct.

     version=6.3.0-SNAPSHOT
     docker image pull fcrepo/fcrepo:$version
     docker image list fcrepo/fcrepo:$version
     echo -e "\n\n"

     docker run --rm --name=fcrepo --publish 8666:8080 fcrepo/fcrepo:latest 2>fcrepo.err >fcrepo.log &
     sleep 5

     info_fedora='http://localhost:8666/fcrepo/rest'
     ttl='
         @prefix ldp:   <http://www.w3.org/ns/ldp#> .
         @prefix schema: <http://schema.org/> .
         @prefix wiki: <http://en.wikipedia.org/> .
         <>  schema:name "Ashley";
             schema:about wiki:Cat;
              .

         wiki:Cat schema:name "Cat".
         ';

     ashley='
          \    /\
           )  ( `)
          (  /  )
           \(__)|
     ';
     curl -XPUT -u fedoraAdmin:fedoraAdmin -H'Content-type:text/turtle' --data-binary "@-" $info_fedora/ashley <<< "$ttl"
     curl -XPUT -u fedoraAdmin:fedoraAdmin -H'Content-type:text/x-ascii-art' --data-binary "@-" $info_fedora/ashley.art <<< "$ashley"
     curl -XPUT -u fedoraAdmin:fedoraAdmin -H'Content-type:text/turtle' --data-binary "@-" $info_fedora/ashley.art/fcr:metadata <<< "$ttl"

     echo -e "\n========= LDP ========\n"
     for i in ashley.art ashley.art/fcr:metadata ashley; do
         echo "========== $i =========="
         curl -u fedoraAdmin:fedoraAdmin -H'PREFER:return=representation; omit="http://fedora.info/definitions/fcrepo#ServerManaged"' $info_fedora/$i
         echo -e "========== $i ==========\n"

     done
     echo -e "\n========= OCFL ========\n"
     echo "========== ashley.art~fcr-desc.nt =========="
     docker exec fcrepo cat fcrepo-home/data/ocfl-root/889/c1a/e3f/889c1ae3f46c7952fd40cd139ee5a6225654f659913aba5d2de2533c1841d3fa/v2/content/ashley.art~fcr-desc.nt
     echo -e "========== ashley.art~fcr-desc.nt ==========\n"

     docker stop fcrepo > /dev/null
