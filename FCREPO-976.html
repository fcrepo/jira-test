This is resolved in FgsConfig/FgsConfigIndexTemplate/Solr/foxmlToSolr.xslt by

<xsl:template match="/foxml:digitalObject" mode="inactiveFedoraObject">
<delete>
<id><xsl:value-of select="$PID"/></id>
</delete>
</xsl:template>
