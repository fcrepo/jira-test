Note: When implementing this, the integration-tests will need the following configuration:

ExampleIT.java

        client = FcrepoClient.client().credentials("fedoraAdmin", "password").authScope("localhost").build();

web.xml

   <!--filter to add test auth to grizzly -->
  <filter>
    <filter-name>TestAuth</filter-name>
    <filter-class&gt;org.fcrepo.http.commons.test.util.TestAuthenticationRequestFilter</filter-class&gt;
  </filter>

  <filter-mapping>
    <filter-name>TestAuth</filter-name>
    <url-pattern>/rest/*</url-pattern>
  </filter-mapping>

test-repository.json

    "security" : {
        "anonymous" : {
            "roles" : ["readonly","readwrite","admin"],
            "useOnFailedLogin" : false
        },
        "providers" : [
            { "classname" : "org.fcrepo.auth.common.ServletContainerAuthenticationProvider" }
        ]
    },
