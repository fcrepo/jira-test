@Test
public void testAddDatastream()
{
	log.info("[ testAddDatastream ]");

	String filename = "data/test-datastream01.xml";
	try
	{
		String dsID = "TESTDS";
		String[] altIDs = new String[0];
		String dsLabel = "Test Datastream";
		boolean versionable = true;
		String mimeType = "text/xml";
		String formatURI = null;
		String dsLocation = fedora.uploadFile(new File(filename));
		String controlGroup = "X";
		String dsState = "A";
		String checksumType = "MD5";
		String doc = new String(DataUtils.loadFile(filename), "UTF-8");
		String checksum = DataUtils.getHash(DataUtils.fedoraXMLHashFormat(doc.getBytes("UTF-8")));
		String logMessage = "test add datastream";

		String dsidResult = apim.addDatastream(pid, dsID, altIDs, dsLabel, versionable, mimeType, formatURI,
			dsLocation, controlGroup, dsState, checksumType, checksum, logMessage);

		if (log.isDebugEnabled())
			log.debug("Returned Datastream ID: " + dsidResult);

		Assert.assertEquals("Unexpected result from addDatastream", dsID, dsidResult);

		Datastream ds = apim.getDatastream(pid, dsID, null);
		if (log.isDebugEnabled())
			log.debug("[ Datastream ] " + ds.getID() + ": " + ds.getLabel() + " [ " + ds.getLocation() + " ]");

		String actual = ds.getChecksum();

		if (log.isDebugEnabled())
		{
			log.debug("actual checksum:   " + actual);
			log.debug("expected checksum: " + checksum);
		}

		Assert.assertEquals("Actual != expected", checksum, actual);
	}
	catch (Exception e)
	{
		Assert.fail(e.getMessage());
	}
}
