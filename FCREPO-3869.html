FCREPO-3869 '+' character in URI identifier causes issues with `Accept:text/html` GET requests

Description

Adding the '+' character in URI identifiers seems to work fine, and properly decodes for PUTs and GETs for most Accept mime-types, but seems to mess up
the http interface to fcrepo. The URI seems to be decoded twice, converting
the '+' as shorthand for ' '. Other http encoded characters to not seem to be affected.

      version=6.3.0
      docker image pull fcrepo/fcrepo:$version
      docker run --rm --name=fcrepo --publish 8666:8080 fcrepo/fcrepo:${version} 2>fcrepo.err >fcrepo.log &
      sleep 5
     info_fedora="http://localhost:8666/fcrepo/rest"
     put="curl -XPUT -u fedoraAdmin:fedoraAdmin -H'Content-Type:application/ld+json'"
     
     for n in 'c&h' 'c+h' 'c_h'; do
         encoded=$(jq -sRr --arg x $n '$x|@uri');
         v=$(echo -n "info:fedora/$n" | shasum -a 256); echo $n $encoded $v;
         curl -XPUT -u fedoraAdmin:fedoraAdmin -H'Content-Type:application/ld+json' --data "{\"@id\":\"\",\"http://schema.org/name\":\"$n\"}" ${info_fedora}/$encoded > /dev/null
     done
     
     for n in 'c&h' 'c+h' 'c_h'; do
         encoded=$(jq -sRr --arg x $n '$x|@uri');
         echo "=$n or $encoded ="
         curl -XGET -u fedoraAdmin:fedoraAdmin -H'Accept:application/ld+json' --head ${info_fedora}/$encoded | grep HTTP
         curl -XGET -u fedoraAdmin:fedoraAdmin -H'Accept:application/ld+json' ${info_fedora}/$encoded | jq -r '[.[]["@id"],.[]["http://schema.org/name"][]["@value"]]'
         echo '==html=='
         curl -XGET -u fedoraAdmin:fedoraAdmin -H'Accept:text/html' --head ${info_fedora}/$encoded | grep HTTP
     done
     docker stop fcrepo
